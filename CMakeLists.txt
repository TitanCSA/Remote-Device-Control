cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(RemoteDeviceControl VERSION 0.3.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 Desktop C++.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(RDC_ASAN "Instrument our user-mode tests with AddressSanitizer" OFF)
set(RDC_UPSTREAM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/obj/Upstream" CACHE PATH "Pinned Moonlight and MbedTLS source; see Tools/Prepare-Moonlight.py")
if(NOT EXISTS "${RDC_UPSTREAM_DIR}/MoonlightCommon/CMakeLists.txt" OR NOT EXISTS "${RDC_UPSTREAM_DIR}/MbedTLS/CMakeLists.txt")
 message(FATAL_ERROR "Run python Tools/Prepare-Moonlight.py before configuring.")
endif()
# Keep upstream warning policy independent from the strict native app policy.
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/Source/MbedConfig.h" CACHE FILEPATH "Windows threading integration" FORCE)
add_subdirectory("${RDC_UPSTREAM_DIR}/MbedTLS" dependencies/mbedtls)
foreach(cryptoTarget IN ITEMS mbedcrypto mbedx509 mbedtls)
 target_include_directories(${cryptoTarget} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Source")
endforeach()
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
add_subdirectory("${RDC_UPSTREAM_DIR}/MoonlightCommon" dependencies/moonlight)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(RDC_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(RemoteTransport Source/Transport.cpp)
target_link_libraries(RemoteTransport PUBLIC ws2_32 secur32 crypt32 ncrypt bcrypt)
target_include_directories(RemoteTransport PUBLIC Source)
add_library(RemoteCommon Source/Common.cpp Source/Config.cpp Source/Discovery.cpp)
target_link_libraries(RemoteCommon PUBLIC RemoteTransport advapi32 shell32 ole32 iphlpapi uuid)
add_library(RemoteInput Source/VirtualInput.cpp Vdevice/Vdevice.cpp)
set_source_files_properties(Vdevice/Vdevice.cpp PROPERTIES COMPILE_OPTIONS "/wd4005")
target_link_libraries(RemoteInput PUBLIC RemoteCommon cfgmgr32)
add_library(RemoteVideo Source/GameStreamCodec.cpp)
target_link_libraries(RemoteVideo PUBLIC RemoteCommon mfplat mf mfuuid d3d11 dxgi evr ole32 oleaut32 strmiids wmcodecdspuuid)
add_library(RemoteGameStream Source/Client.cpp Source/GameStreamClient.cpp Source/MbedThreading.cpp)
target_link_libraries(RemoteGameStream PUBLIC RemoteVideo RemoteInput moonlight-common-c mbedcrypto)
add_executable(RemoteDeviceHost WIN32 Source/HostMain.cpp Source/Host.cpp Source/CaptureAgent.cpp)
target_link_libraries(RemoteDeviceHost PRIVATE RemoteCommon wtsapi32 userenv shell32 comctl32)
target_link_options(RemoteDeviceHost PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(RemoteDeviceControl WIN32 Source/ViewerMain.cpp)
target_link_libraries(RemoteDeviceControl PRIVATE RemoteGameStream d2d1 comctl32)
target_link_options(RemoteDeviceControl PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_library(RemoteSetup Source/Setup.cpp Source/EngineBundle.cpp)
target_link_libraries(RemoteSetup PUBLIC RemoteInput wintrust crypt32 ole32 oleaut32 setupapi uuid user32 gdi32 uxtheme cabinet)
set(VDEVICE_SETUP_FILE "" CACHE FILEPATH "Unchanged Microsoft-signed Vdevice demo console installer")
set(RDC_ENGINE_CAB "" CACHE FILEPATH "Reproducibly built Sunshine engine cabinet")
set(RDC_ENGINE_MANIFEST "" CACHE FILEPATH "RDC-ENGINE-1 cabinet file manifest")
if(VDEVICE_SETUP_FILE AND RDC_ENGINE_CAB AND RDC_ENGINE_MANIFEST)
 foreach(payload IN ITEMS VDEVICE_SETUP_FILE RDC_ENGINE_CAB RDC_ENGINE_MANIFEST)
  if(NOT EXISTS "${${payload}}")
   message(FATAL_ERROR "Missing ${payload}")
  endif()
 endforeach()
 file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/Uninstaller-$<CONFIG>.rc" CONTENT "301 RCDATA \"${VDEVICE_SETUP_FILE}\"\n302 RCDATA \"$<TARGET_FILE:RemoteDeviceHost>\"\n")
 file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/Installer-$<CONFIG>.rc" CONTENT "301 RCDATA \"${VDEVICE_SETUP_FILE}\"\n302 RCDATA \"$<TARGET_FILE:RemoteDeviceHost>\"\n303 RCDATA \"$<TARGET_FILE:RemoteDeviceControlUninstall>\"\n304 RCDATA \"${RDC_ENGINE_CAB}\"\n305 RCDATA \"${RDC_ENGINE_MANIFEST}\"\n")
 add_executable(RemoteDeviceControlUninstall WIN32 Source/InstallerMain.cpp "${CMAKE_BINARY_DIR}/Uninstaller-$<CONFIG>.rc")
 target_compile_definitions(RemoteDeviceControlUninstall PRIVATE RDC_UNINSTALL_ONLY)
 target_link_libraries(RemoteDeviceControlUninstall PRIVATE RemoteSetup comctl32)
 target_link_options(RemoteDeviceControlUninstall PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
 add_dependencies(RemoteDeviceControlUninstall RemoteDeviceHost)
 add_executable(RemoteDeviceControlSetup WIN32 Source/InstallerMain.cpp "${CMAKE_BINARY_DIR}/Installer-$<CONFIG>.rc")
 add_dependencies(RemoteDeviceControlSetup RemoteDeviceHost RemoteDeviceControlUninstall)
 target_link_libraries(RemoteDeviceControlSetup PRIVATE RemoteSetup comctl32)
 target_link_options(RemoteDeviceControlSetup PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endif()
enable_testing()
foreach(test IN ITEMS Core Input Discovery Setup GameStream GameStreamCodec EngineBundle)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/${test}Tests.cpp")
  add_executable(${test}Tests Tests/${test}Tests.cpp)
  if(test STREQUAL "Input")
   target_link_libraries(${test}Tests PRIVATE RemoteInput)
  elseif(test STREQUAL "Setup" OR test STREQUAL "EngineBundle")
   target_link_libraries(${test}Tests PRIVATE RemoteSetup)
  elseif(test STREQUAL "GameStream")
   target_link_libraries(${test}Tests PRIVATE RemoteGameStream mbedcrypto)
  elseif(test STREQUAL "GameStreamCodec")
   target_link_libraries(${test}Tests PRIVATE RemoteVideo)
  else()
   target_link_libraries(${test}Tests PRIVATE RemoteCommon)
  endif()
  add_test(NAME ${test}Tests COMMAND ${test}Tests)
  set_tests_properties(${test}Tests PROPERTIES TIMEOUT 60)
 endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/Observer.cpp")
 add_executable(LabObserver EXCLUDE_FROM_ALL WIN32 Tests/Observer.cpp)
 target_link_libraries(LabObserver PRIVATE user32 gdi32)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/LabPairing.cpp")
 add_executable(LabPairing EXCLUDE_FROM_ALL Tests/LabPairing.cpp)
 target_link_libraries(LabPairing PRIVATE RemoteSetup)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/obj/DirectLiveProbe.cpp")
 add_executable(DirectLiveProbe EXCLUDE_FROM_ALL obj/DirectLiveProbe.cpp)
 target_link_libraries(DirectLiveProbe PRIVATE RemoteGameStream)
endif()
