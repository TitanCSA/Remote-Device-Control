cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(RemoteDeviceControl VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 Desktop C++.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(RDC_ASAN "Instrument user-mode tests with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(RDC_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(RemoteTransport Source/Transport.cpp Source/RelayStream.cpp)
target_link_libraries(RemoteTransport PUBLIC ws2_32 secur32 crypt32 ncrypt bcrypt winhttp)
target_include_directories(RemoteTransport PUBLIC Source)
add_library(RemoteCommon Source/Common.cpp Source/Config.cpp)
target_link_libraries(RemoteCommon PUBLIC RemoteTransport advapi32 shell32 ole32 iphlpapi)
add_library(RemoteInput Source/VirtualInput.cpp Vdevice/Vdevice.cpp)
# The unchanged two-file upstream SDK defines this macro without a value.
set_source_files_properties(Vdevice/Vdevice.cpp PROPERTIES COMPILE_OPTIONS "/wd4005")
target_link_libraries(RemoteInput PUBLIC RemoteCommon cfgmgr32)
add_library(RemoteCapture Source/Capture.cpp Source/FrameCodec.cpp)
target_link_libraries(RemoteCapture PUBLIC RemoteCommon d3d11 dxgi windowscodecs user32 gdi32 ole32)
add_executable(RemoteDeviceHost WIN32 Source/HostMain.cpp Source/Host.cpp Source/CaptureAgent.cpp)
target_link_libraries(RemoteDeviceHost PRIVATE RemoteInput RemoteCapture wtsapi32 userenv shell32 comctl32)
target_link_options(RemoteDeviceHost PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(RemoteDeviceControl WIN32 Source/ViewerMain.cpp Source/Client.cpp)
target_link_libraries(RemoteDeviceControl PRIVATE RemoteInput RemoteCapture d2d1 comctl32)
target_link_options(RemoteDeviceControl PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Source/Setup.cpp")
 set(VDEVICE_SETUP_FILE "" CACHE FILEPATH "Unchanged matching Microsoft-signed Vdevice demo console installer")
 if(VDEVICE_SETUP_FILE AND EXISTS "${VDEVICE_SETUP_FILE}")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/Uninstaller-$<CONFIG>.rc" CONTENT "301 RCDATA \"${VDEVICE_SETUP_FILE}\"\n302 RCDATA \"$<TARGET_FILE:RemoteDeviceHost>\"\n")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/Installer-$<CONFIG>.rc" CONTENT "301 RCDATA \"${VDEVICE_SETUP_FILE}\"\n302 RCDATA \"$<TARGET_FILE:RemoteDeviceHost>\"\n303 RCDATA \"$<TARGET_FILE:RemoteDeviceControlUninstall>\"\n")
  add_library(RemoteSetup Source/Setup.cpp)
  target_link_libraries(RemoteSetup PUBLIC RemoteInput wintrust crypt32 ole32 oleaut32 setupapi uuid user32 gdi32 uxtheme)
  add_executable(RemoteDeviceControlUninstall WIN32 Source/InstallerMain.cpp "${CMAKE_BINARY_DIR}/Uninstaller-$<CONFIG>.rc")
  target_compile_definitions(RemoteDeviceControlUninstall PRIVATE RDC_UNINSTALL_ONLY)
  target_link_libraries(RemoteDeviceControlUninstall PRIVATE RemoteSetup comctl32)
  target_link_options(RemoteDeviceControlUninstall PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
  add_dependencies(RemoteDeviceControlUninstall RemoteDeviceHost)
  add_executable(RemoteDeviceControlSetup WIN32 Source/InstallerMain.cpp "${CMAKE_BINARY_DIR}/Installer-$<CONFIG>.rc")
  add_dependencies(RemoteDeviceControlSetup RemoteDeviceHost RemoteDeviceControlUninstall)
  target_link_libraries(RemoteDeviceControlSetup PRIVATE RemoteSetup comctl32)
  target_link_options(RemoteDeviceControlSetup PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/SetupTests.cpp")
   add_executable(SetupTests Tests/SetupTests.cpp)
   target_link_libraries(SetupTests PRIVATE RemoteSetup)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/LabPairing.cpp")
   add_executable(LabPairing EXCLUDE_FROM_ALL Tests/LabPairing.cpp)
   target_link_libraries(LabPairing PRIVATE RemoteSetup)
  endif()
 endif()
endif()
enable_testing()
add_executable(CoreTests Tests/CoreTests.cpp)
target_link_libraries(CoreTests PRIVATE RemoteCommon)
add_test(NAME PinnedTlsPersistedIdentityAndProtocol COMMAND CoreTests)
set_tests_properties(PinnedTlsPersistedIdentityAndProtocol PROPERTIES TIMEOUT 40)
add_executable(InputTests Tests/InputTests.cpp)
target_link_libraries(InputTests PRIVATE RemoteInput)
add_test(NAME VirtualInputDescriptorsAndState COMMAND InputTests)
add_executable(CaptureTests Tests/CaptureTests.cpp)
target_link_libraries(CaptureTests PRIVATE RemoteCapture)
add_test(NAME JpegBoundsAndRoundtrip COMMAND CaptureTests)
if(TARGET SetupTests)
 add_test(NAME InstallerOwnershipAndArguments COMMAND SetupTests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/Observer.cpp")
 add_executable(LabObserver EXCLUDE_FROM_ALL WIN32 Tests/Observer.cpp)
 target_link_libraries(LabObserver PRIVATE user32 gdi32)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/LiveProbe.cpp")
 add_executable(LiveProbe EXCLUDE_FROM_ALL Tests/LiveProbe.cpp Source/Client.cpp)
 target_link_libraries(LiveProbe PRIVATE RemoteInput RemoteCapture)
endif()
